cmake_minimum_required(VERSION 3.22)
project(mogo_mobile_jni LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MOGO_ENABLE_HTTP OFF CACHE BOOL "" FORCE)
set(MOGO_ENABLE_NATIVE ON CACHE BOOL "" FORCE)
set(MOGO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../../backend" mogo-backend)
add_library(mogo_mobile_jni SHARED src/main/cpp/mogo_jni.cpp)
target_link_libraries(mogo_mobile_jni PRIVATE mogo_native_core)
